rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /salas/{salaId} {
      allow read, write: if request.auth != null;
      match /participantes/{participanteId} { allow read, write: if request.auth != null; }
      match /respuestas/{respuestaId} { allow read, write: if request.auth != null; }
      match /lecturasResultado/{confirmacionId} {
        allow read: if request.auth != null;
        allow create, update: if request.auth != null && request.resource.data.uid == request.auth.uid;
        allow delete: if request.auth != null;
      }
      match /revancha/{usuarioId} {
        allow read: if request.auth != null;
        allow create, update: if request.auth != null && request.resource.data.uid == request.auth.uid && usuarioId == request.auth.uid;
        allow delete: if request.auth != null && usuarioId == request.auth.uid;
      }
    }
    match /usuarios/{usuarioId} {
      allow read, create, update, delete: if request.auth != null && request.auth.uid == usuarioId;
      match /{documento=**} {
        allow read, create, update, delete: if request.auth != null && request.auth.uid == usuarioId;
      }
    }
  }
}
